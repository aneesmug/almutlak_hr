 * ============================================================================
 * DATABASE QUERIES CHEAT SHEET
 * ============================================================================
 * 
 * Fetch all comments for a request:
 *   SELECT * FROM approval_comments 
 *   WHERE request_inv_no = 'VAC-2025-001' 
 *   AND request_type = 'vacation'
 *   ORDER BY comment_date ASC;
 * 
 * Get latest comment:
 *   SELECT * FROM approval_comments 
 *   WHERE request_inv_no = 'VAC-2025-001'
 *   ORDER BY comment_date DESC LIMIT 1;
 * 
 * Count by action:
 *   SELECT approval_action, COUNT(*) as count
 *   FROM approval_comments
 *   WHERE request_inv_no = 'VAC-2025-001'
 *   GROUP BY approval_action;
 * 
 * Find all rejections with reasons:
 *   SELECT * FROM approval_comments
 *   WHERE approval_action = 'rejected'
 *   ORDER BY comment_date DESC;
 * 
 * Get approval timeline:
 *   SELECT approver_name, approval_action, approval_level, 
 *          comment_text, comment_date
 *   FROM approval_comments
 *   WHERE request_inv_no = 'VAC-2025-001'
 *   ORDER BY approval_level, comment_date;
